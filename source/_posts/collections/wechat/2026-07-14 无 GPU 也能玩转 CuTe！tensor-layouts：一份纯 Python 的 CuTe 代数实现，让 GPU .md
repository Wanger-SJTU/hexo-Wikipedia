把多个 Layout "捆绑"成新 Layout 的不同 mode            shapes  = tuple(L.shape  for L in args)            strides = tuple(L.stride for L in args)            self._shape, self._stride = shapes, strides        elif len(args) == 1:            # 只给 shape，按列主序自动生成 stride            self._shape  = normalize(args[0])            self._stride = compute_col_major_strides(self._shape)        elif len(args) == 2:            shape, stride = args            self._shape  = normalize(shape)            self._stride = normalize(stride)        # congruent: shape 和 stride 的嵌套结构必须完全同构        ifnot congruent(self._shape, self._stride):            raise LayoutError(...)    def __call__(self, *args):        """coord -> offset，或 None 触发切片。"""        coords = args[0] if len(args) == 1else args        if has_none(coords):              # 切片：返回子 Layout            return Layout(slice_modes(coords, self.shape),                          slice_modes(coords, self.stride))        return crd2offset(coords, self.shape, self.stride)注意上面对 congruent 约束——这是 CuTe 的核心不变量：shape 与 stride 必须有相同的层次结构。这条约束让所有递归算法（fold、zip_transform、transform_tuple）都能"按 shape 走、stride 同步走"，不必担心结构错位。底层的 crd2offset 把"坐标 → 偏移"的语义讲得最为透彻：# 来源：src/tensor_layouts/layouts/algebra.pydef crd2offset(coord, shape, stride) -> int:    if is_int(coord):                          # 一维 flat index → 多维        offset = 0        index = coord        for i, (s, d) in enumerate(zip(shape, stride)):            mode_size = size(s)            if i < len(shape) - 1:             # 非最后维：取模                c = index % mode_size                index //= mode_size            else:                              # 最后一维不取模——可"无限延伸"                c = index            if is_tuple(s):                offset += crd2offset(c, s, d)  # 嵌套递归            else:                offset += c * d        return offset    # ...nD coordinate 路径同理关键细节是"最后一维不取模"。这条 CuTe 公约让张量在最外层维度上是"隐式无限"的——配合 logical_divide 切 tile 时极其方便，因为你不必关心被切对象的具体长度。3.2 Swizzle：XOR 实现的非线性变换GPU 共享内存被分成 32 个 bank，多个线程同时命中同一 bank 会被串行化（bank conflict）。CuTe 用一个位异或函数绕过冲突；它无法用 stride 表示，因此被独立建模：# 来源：src/tensor_layouts/layouts/core.pyclass Swizzle:    """Swizzle(bits, base, shift):       index bit pattern 0bxxxYYYxxxxZZZxxxx       - base : 保持不变的低位个数 (rightmost xxxx)       - bits : YYY 与 ZZZ 的位宽       - shift: YYY 与 ZZZ 之间的距离       结果：将 ZZZ 替换为 (ZZZ XOR YYY)。"""    def __call__(self, idx: int) -> int:        mask = ((1 << self.bits) - 1) << self.base        if self.shift >= 0:            return idx ^ ((idx >> self.shift) & mask)        return idx ^ ((idx & mask) << (-self.shift))举一个具体例子（来自源码文档）：Swizzle(3, 0, 3) 把第 [0,3) 位与第 [3,6) 位异或：index 19 = 0b010_011  (Y=010=2, Z=011=3)       → 0b010_(011 XOR 010) = 0b010_001 = 17相邻行因此被映射到不同的 bank。make_swizzle(Y, Z) 工具函数会反向构造 Swizzle，并在两个掩码的位计数不匹配或区域重叠时直接抛 LayoutError——这种"宁可拒绝构造、不允许产生半正确对象"的防御式编程贯穿了全项目。3.3 ComposedLayout：装下"非仿射"的容器Layout 本身严格仿射（即 offset = ∑ coord_i · stride_i）。但 Swizzle 是非线性的，无法揉进 stride。项目的解法是引入一个独立的表达式节点：# 来源：src/tensor_layouts/layouts/expr.py@dataclass(frozen=True)class ComposedLayout:    """Semantics:        ComposedLayout(outer, inner, offset)(coord)            == outer(offset + inner(coord))    """    outer: Any     # 可以是 Swizzle 或 Layout    inner: Any     # 通常是 Layout，逆向运算时也可为 Swizzle    offset: int = 0文件头注释明确指出这是项目的"Path X"重构——历史上曾把 swizzle 直接嵌入 Layout 类，后来发现可读性与正确性都吃亏，于是把所有非仿射性质都隔离到 ComposedLayout。这种"汇聚复杂性、保护核心 Layout 简洁"的边界选择，是项目最值得借鉴的设计哲学。unsetunset四、代数运算的内核实现unsetunset4.1 compose：函数复合的工业级算法compose(A, B) 的语义是 C(i) = A(B(i))——B 决定"以什么顺序访问 A 的哪些元素"。algebra.py 用 dispatcher 模式按 (lhs, rhs) 的类型组合分发：# 来源：src/tensor_layouts/layouts/algebra.pydef compose(layout_a, layout_b):    # LHS 解包    if isinstance(layout_a, ComposedLayout):        return _compose_into_composed_lhs(layout_a, layout_b)    if isinstance(layout_a, Swizzle):        return _compose_swizzle_lhs(layout_a, layout_b)    # 到这里 LHS 一定是纯仿射 Layout    if isinstance(layout_b, Swizzle):        return _compose_with_swizzle_rhs(layout_a, layout_b)    if isinstance(layout_b, Layout):        return _compose_layout_with_layout(layout_a, layout_b)    if is_tuple(layout_b):        return _compose_with_tuple_tiler(layout_a, layout_b)    # ...最有趣的是底层的一维复合 _composition_1d，它给出了 CuTe 论文里那个绕来绕去的"折叠算法"的清晰 Python 复现：# 来源：src/tensor_layouts/layouts/algebra.pydef _composition_1d(layout_a, b_shape, b_stride):    """compose A with a 1D layout (b_shape:b_stride).       1. 把 A coalesce 到扁平模式列表。       2. 在 A 的每个 mode（除最后一个）上：          - 计算这个 mode 能"消化"多少 B 的元素          - 产出一个 (consumed_shape, b_stride * a_stride) 的输出 mode          - 把剩余的 B 形状/步长传到下一个 mode       3. 最后一个 mode 吸收所有剩余形状（CuTe 的"最外维隐式无限"约定）。"""    flat_a = coalesce(layout_a)    remaining_shape, remaining_stride = b_shape, b_stride    result_shape, result_stride = [], []    for curr_shape, curr_stride in zip(flat_a.shape[:-1], flat_a.stride[:-1]):        divisible  = curr_shape % remaining_stride == 0 \                  or remaining_stride % curr_shape == 0        fits_in_mode = (remaining_shape - 1) * remaining_stride < curr_shape        if fits_in_mode:            # B 完全装进当前 A mode，结束            result_shape.append(remaining_shape)            result_stride.append(remaining_stride * curr_stride)            remaining_shape = 1            break        # 否则消耗这个 mode，进位到下一个        ...    # 最外维兜底    if remaining_shape != 1:        result_shape.append(remaining_shape)        result_stride.append(remaining_stride * flat_a.stride[-1])    return Layout(as_shape(result_shape), as_shape(result_stride))整段代码可以理解为"像做长除法一样，把 B 的访问模式逐位映射到 A 的层级 shape 上"。每跨过一个 A 的 mode，就把"消耗"和"进位"分别记录到 result 与 remaining 里——和小学生做的笔算除法在结构上完全一致。4.2 complement：填补 stride 间隙的折叠complement(L) 的直觉是："L 跳过了哪些偏移，就让 complement 把它们顺序走完。"两者合起来 Layout(L, complement(L)) 是 [0, cosize) 的一个双射。# 来源：src/tensor_layouts/layouts/algebra.pydef complement(layout, cosize_bound=None):    """E.g. complement(Layout(4, 2), 16) -> Layout((2,2), (1,8)).       L = 4:2 → 访问 {0,2,4,6}       complement 必须填进每个 stride-2 间隙的 stride-1 元素，       并把剩余空间用更大的 stride 跳过去填到 16。"""    def _step_mode(current_stride, stride, shape):        gap_size = stride // current_stride if stride > current_stride else1        return gap_size, stride * shape    flat = flatten(layout)    modes = sorted((d, s) for s, d in zip(flat.shape, flat.stride)                          if s != 1and d != 0)    result_shapes, result_strides = [], []    current_stride = 1    for stride, shape in modes:           # 按 stride 升序处理        gap_size, next_stride = _step_mode(current_stride, stride, shape)        if gap_size > 1:                  # 有 gap 才需要 emit            result_shapes.append(gap_size)            result_strides.append(current_stride)        current_stride = next_stride    # 用最后一个尾部模式填到 cosize_bound    remaining = _ceil_div(cosize_bound, current_stride)    result_shapes.append(remaining)    result_strides.append(current_stride)    return coalesce(Layout(as_shape(result_shapes), as_shape(result_strides)))它就像在数轴上排"砖头"——L 的每个 mode 占据一些 stride 间隔，complement 把剩下的孔洞按从小到大顺序敲入大小不一的砖块，最后再把超出最大 mode 之外的尾巴一次补齐。4.3 logical_divide：用 complement 表达"分块"有了 complement，logical_divide 的实现几乎是一句翻译公式：# 来源：src/tensor_layouts/layouts/algebra.pydef logical_divide(layout, tiler):    """logical_divide(A, B) =         compose(A, Layout(B, complement(B, size(coalesce(A)))))"""    if isinstance(tiler, Layout):        tiler_complement = complement(tiler, size(coalesce(layout)))        combined = Layout(tiler, tiler_complement)   # (tile, rest)        return compose(layout, combined)             # 用 A 取址    ...物理含义在源码注释里讲得很形象：要把 A 切成大小为 B 的 tile，你需要两个坐标——"tile 内的位置"（用 B 描述）与"哪一片 tile"（用 complement(B) 描述）。把它们捆成一个二维坐标空间 Layout(B, complement(B))，再用 A 把这套坐标翻译成内存偏移，就完成了分块。执行结果举例：# 来源：tests/paper_examples.pyA = Layout(24, 3)          # shape=24, stride=3 → 访问 {0,3,6,...,69}B = Layout(8, 3)R = logical_divide(A, B)# R = Layout((8, 3), (9, 3))# 同一组偏移，但访问顺序按 B 的 stride 重排只用 4 行核心代码，就把"图像切 tile、矩阵分块乘"的硬件级抽象一次性形式化了。unsetunset五、与硬件相遇：MMA Atom 与 Swizzle 可视化unsetunset5.1 MMA 原子：四家厂商一张表atoms_nv.py 中定义了从 SM61 (Pascal DP4A) 到 SM120 (Blackwell B200)、甚至 GMMA warpgroup 与 sparse 变体的全部布局。每个原子都是一个 dataclass：# 来源：src/tensor_layouts/atoms.py@dataclass(frozen=True)class MMAAtom:    name: str    ptx: str               #
