### FP16 转 FP32

给定一个FP16 x" role="presentation">xx `{S|EEEEE|MMMMM MMMMM}`，要求计算FP32 y=x。我们首先用位运算构造FP32 y' `{S|000EEEEE|MMMMM MMMMM 00000 00000 000}`。 y' 有一个特别美好的性质，如果 x 是Denorm的（`EEEEE==0`），那么 y' 也是Denorm的，而且 y' = 2^{-112} * x ，可以看下面这两个例子。

x=0x3EAA
0|    01111|10101 01010
  2**(0) * 1.666

y1=0x07D54000
0|000 01111|10101 01010 00000 00000 000
  2**(-112) * 1.666

x=0x0155
0|    00000|01010 10101
  2**(-14) * 0.333

y1=0x002AA000
0|000 00000|01010 10101 00000 00000 000
  2**(-126) * 0.333

接着只要一次fp32乘法，我们就能找到y了。NAN和±INF的情况还是需要额外的if branch。

### FP32 转 FP16

Downcast要比Upcast麻烦的多，因为有个Rounding。IEEE定义了一些Rounding方法，比较常用的（也是最烦的）是Round Towards Nearest, tie to Even (RTNE)。放到FP32转FP16里就是，FP32比FP16多出来的13个Mantissa，如果大于`10000...`就在对齐FP16的最后一个Mantissa Bit上+1。如果等于`10000...`（tie），那么只有当对齐FP16的最后一个Mantissa Bit是1时，才会在这个Bit上+1。这样最后Rounding出来的数，最后一位是0（even）。

注意到乘法指令自带一次Rounding。给定 y ，我们先乘一个 2^{-ebits-sth} ，构造出 y'' =`{S|00000000|00000 00000 000MM MMMMM MMM}`。乘法指令的Rounding作用在我们留下的10个MBits上，相当于完成了Downcast的Rounding。我们再乘一个 1">2^{-112+ebits+sth}>1 ，把 y'' 还原成 y' 。如果FP32 y对应的FP16 x是Denorm的，那么我们在 y''也需要更多的0，代码里`delta`就在算这个。

FP16精度范围是 2^{-14} * (0 + 2^{-10}) 到 2^{14} * (1+...) 。对应到FP32 y，ebits最大到`0b10001101=141`。也就是说我们乘的第一个 2^{-ebits-sth} 在FP32下会下溢到0。所以我们把乘法分成两次，第一次固定乘 2^{-100} ，第二次再乘 2^{100-ebits-sth} 。对于FP16的精度范围，第一次乘 2^{-100} 在FP32下必定不会有Rounding。超出精度范围的值，有if branch兜底。

### FP32 转 BF16

FP32转BF16的Rounding，我们可以在Int模式下加一个固定的数`0b 1 00000 00000 0000 - !bit[17]`。神奇的是，Rounding一路进位到了指数位上，也是正确的行为。Ebits变成了`111 11111`的情况，也刚好是上溢到INF的情况，我们可以把判INF的If Branch给丢掉。（NAN的Branch还得留，因为区别INF和Nan的1，可能全在后16个MBits上）

constexpr float normal_pow2(int x) {
    DEBUG(assert(x > -126));
    auto ebits = x + 127;
    return std::bit_castfloat>(ebits  23);
}

bool zzy_isnan(uint16_t f16_bits) {
    uint16_t mbits = f16_bits & (0b11111'11111);
    uint16_t ebits = (f16_bits >> 10) & (0b11111);
    return ebits == 0b11111 && mbits > 0;
}

bool zzy_isnan1(uint16_t f16_bits) {
    auto z = uint32_t(f16_bits)  13;
    auto z1 = z | 0xF0000000;
    auto z2 = std::bit_castfloat>(z1);
    return isnan(z2);
}

float zzy_upcast(uint16_t f16_bits) {
    if (f16_bits == 0x7C00) {
        return std::numeric_limitsfloat>::infinity();
    }
    if (f16_bits == 0xFC00) {
        return -std::numeric_limitsfloat>::infinity();
    }
    if (zzy_isnan(f16_bits)) {
        return std::numeric_limitsfloat>::quiet_NaN();
    }

    uint32_t y1_bits = (uint32_t(f16_bits)  17) >> 4;
    DEBUG(printf("%08x\n", std::bit_castuint32_t>(y1_bits)));
    y1_bits |= (uint32_t(f16_bits) >> 15)  31;
    DEBUG(printf("%08x\n", std::bit_castuint32_t>(y1_bits)));
    float y1 = std::bit_castfloat>(y1_bits);
    float ans = y1 * normal_pow2(112);
    return ans;
}

uint16_t zzy_downcast(float f32) {
    if (f32 >= 65520.0) {
        return 0x7C00;
    }
    if (f32  -65520.0) {
        return 0xFC00;
    }
    if (std::isnan(f32)) {
        return 0x7E00;
    }

    int32_t f32_bits = std::bit_castuint32_t>(f32);
    int32_t ebits = (f32_bits >> 23) & (0b111'11111);

    auto step1 = 100;
    // how many more mantissa bits need to be rounded
    auto delta = std::max((-14 - (ebits - 127)), 0);
    DEBUG(printf("%d\n", delta));
    auto y1 = f32 * normal_pow2(-step1);
    DEBUG(printf("%08x\n", std::bit_castuint32_t>(y1)));
    auto y2 = y1 * normal_pow2(step1 - (ebits + 12 + delta));
    DEBUG(printf("%08x\n", std::bit_castuint32_t>(y2)));
    auto y3 = y2 * normal_pow2((ebits + 12 + delta) - 112);
    DEBUG(printf("%08x\n", std::bit_castuint32_t>(y3)));
    auto bits = std::bit_castint32_t>(y3);

    uint16_t sign = (bits >> 31)  15;
    uint16_t em = (bits >> 13) & 0x7fff;
    uint16_t ans = sign | em;
    return ans;
}

float std_upcast(uint16_t f16_bits) {
    _Float16 f16 = std::bit_cast_Float16>(f16_bits);
    return f16;
}

uint16_t std_downcast(float f32) {
    _Float16 f16 = f32;
    return std::bit_castuint16_t>(f16);
}

void test_upcast() {
    for (uint32_t f16_bits = 0; f16_bits  UINT16_MAX; f16_bits++) {
        if (f16_bits % (1  12) == 0) {
            printf("%01x\n", f16_bits >> 12);
            fflush(stdout);
        }
        auto zzy_f32 = zzy_upcast(f16_bits);
        auto std_f32 = std_upcast(f16_bits);
        auto correct =
            std::isnan(std_f32) ? std::isnan(zzy_f32) : zzy_f32 == std_f32;
        if (!correct) {
            printf("\n\n%04x\n%08x\n%08x", f16_bits,
                   std::bit_castuint32_t>(zzy_f32),
                   std::bit_castuint32_t>(std_f32));
            fflush(stdout);
            throw "GG";
        }
    }
}

void test_downcast() {
    for (uint32_t f32_bits = 0x0  24; f32_bits  UINT32_MAX; f32_bits++) {
        if (f32_bits % (1  24) == 0) {
            printf("%02x\n", f32_bits >> 24);
            fflush(stdout);
        }
        auto f32 = std::bit_castfloat>(f32_bits);
        auto zzy_fp16_bits = zzy_downcast(f32);
        auto std_fp16_bits = std_downcast(f32);
        auto correct = std::isnan(f32) ? zzy_isnan(zzy_fp16_bits)
                                       : zzy_fp16_bits == std_fp16_bits;
        if (!correct) {
            printf("\n\n%08x\n%04x\n%04x", f32_bits, zzy_fp16_bits,
                   std_fp16_bits);
            fflush(stdout);
            throw "GG";
        }
    }
}

__attribute__((noinline)) void view_upcast(uint16_t *__restrict a,
                                           float *__restrict b) {
    for (int i = 0; i  1024; i++) {
        b[i] = zzy_upcast(a[i]);
    }
}

__attribute__((noinline)) void view_downcast(uint16_t *__restrict a,
                                             float *__restrict b) {
    for (int i = 0; i  1024; i++) {
        a[i] = zzy_downcast(b[i]);
    }
}
