### 分析

bbox2offset算子只有一层for循环，不能体现poly优势。在循环轴上无依赖关系，在执行顺序（时间维度）上有依赖关系，所以做循环融合比较有利。

下图是一个依赖关系的简单示例，图中的`Tx`为某条statement，详细内容见代码。

![](/images/collections/0.raw/zhihu/%E4%BD%93%E7%B3%BB%E7%BB%93%E6%9E%84/asset/img_bc4b8f9727a6.jpg)

在PPCG的优化策略里，默认按照tile_size=32进行切分，block_dim = 32, grid_dim = (3000 + 31) / 32。整个for循环轴完全并行。

```cpp
__global__ void kernel0(int *ex_ctr_y, double *ex_heights, int *gt_ctr_y, double *gt_heights, double *offset_dy, double *offset_dh, int *boxes, int *gt, int M)
{
    int b0 = blockIdx.x;
    int t0 = threadIdx.x;
    int private_ex_ctr_y[1];
    double private_ex_heights[1];
    int private_gt_ctr_y[1];
    double private_gt_heights[1];

    {   
      for (int c0 = 32 * b0; c0  M; c0 += 1048576) {
        if (M >= 32 * b0 + t0 + 1 && 32 * b0 + t0  2999 && c0 == 32 * b0) {
          private_ex_ctr_y[0] = ((boxes[(32 * b0 + t0) * 4 + 3] - boxes[(32 * b0 + t0) * 4 + 1]) + 1); 
          private_ex_heights[0] = (boxes[(32 * b0 + t0) * 4 + 1] + (0.5 * private_ex_ctr_y[0]));
          private_gt_ctr_y[0] = ((gt[(32 * b0 + t0) * 4 + 3] - gt[(32 * b0 + t0) * 4 + 1]) + 1); 
          private_gt_heights[0] = (gt[(32 * b0 + t0) * 4 + 1] + (0.5 * private_gt_ctr_y[0]));
        }
        if (M >= t0 + c0 + 1) {
          offset_dh[t0 + c0] = (1.0 * log(private_gt_heights[0] / private_ex_heights[0]));
          if (32 * b0 + t0  2999 && c0 == 32 * b0) {
            offset_dy[32 * b0 + t0] = ((1.0 * (private_gt_ctr_y[0] - private_ex_ctr_y[0])) / private_ex_heights[0]);
            gt_heights[32 * b0 + t0] = private_gt_heights[0];
            ex_heights[32 * b0 + t0] = private_ex_heights[0];
          }
        }
      }   
      if (M >= 32 * b0 + t0 + 1 && 32 * b0 + t0  2999) {
        gt_ctr_y[32 * b0 + t0] = private_gt_ctr_y[0];
        ex_ctr_y[32 * b0 + t0] = private_ex_ctr_y[0];
      }   
    }   
}
```
